Step-by-Step Implementation

HTTP Mode (ngrok)
  • Slack pushes events to a public URL
  • Needs ngrok running to receive events locally
  • Re-paste URL in portal on every ngrok restart
  • Best for production / cloud deployments
Socket Mode
  • Your app opens a WebSocket to Slack
  • No public URL, no ngrok, works behind firewalls
  • Needs one extra App-Level Token (xapp-)
  • Best for local development & testing

const db = require('./database'); db.exec(` CREATE TABLE IF NOT EXISTS tasks ( id TEXT PRIMARY KEY, title TEXT NOT NULL, assignee_id TEXT NOT NULL, created_by TEXT NOT NULL, status TEXT DEFAULT 'OPEN', -- OPEN | COMPLETE | ON_HOLD priority TEXT DEFAULT 'MEDIUM', -- LOW | MEDIUM | HIGH due_date TEXT, channel_id TEXT, created_at TEXT DEFAULT (datetime('now')) ); CREATE TABLE IF NOT EXISTS approvals ( id TEXT PRIMARY KEY, description TEXT NOT NULL, requested_by TEXT NOT NULL, approver_id TEXT NOT NULL, status TEXT DEFAULT 'PENDING', -- PENDING | APPROVED | REJECTED dm_channel_id TEXT, dm_ts TEXT, created_at TEXT DEFAULT (datetime('now')) ); `);
db/tasks.js — function signatures to implement
createTask({ id, title, assigneeId, createdBy, priority, dueDate, channelId }) → db.prepare('INSERT INTO tasks VALUES (?,?,?,?,?,?,?,?)').run(...) getTasksByAssignee(userId) → db.prepare('SELECT * FROM tasks WHERE assignee_id = ?').all(userId) updateTaskStatus(taskId, status) → db.prepare('UPDATE tasks SET status = ? WHERE id = ?').run(status, taskId) getTaskById(taskId) → db.prepare('SELECT * FROM tasks WHERE id = ?').get(taskId)
db/approvals.js — function signatures to implement
createApproval({ id, description, requestedBy, approverId, dmChannelId }) getPendingApprovalsByApprover(userId) → SELECT * FROM approvals WHERE approver_id = ? AND status = 'PENDING' updateApprovalStatus(approvalId, status) ← returns full row updateApprovalDmTs(approvalId, dmTs)

const { v4: uuidv4 } = require('uuid'); const { createTask } = require('../db/tasks'); const { refreshHomeTab } = require('../utils/homeTab'); const { buildTaskCard } = require('../utils/blocks'); module.exports = (app) => { app.view('create_task_modal', async ({ ack, view, body, client }) => { await ack(); // Extract values — path: state.values[block_id][action_id] const vals = view.state.values; const title = vals.task_title_block.task_title_input.value; const assignee = vals.assignee_block.assignee_input.selected_user; const dueDate = vals.due_date_block.due_date_input.selected_date; // null if skipped const priority = vals.priority_block.priority_input.selected_option.value; const channelId = view.private_metadata; const createdBy = body.user.id; const task = { id: uuidv4(), title, assigneeId: assignee, createdBy, priority, dueDate, channelId }; createTask(task); // Post confirmation to channel await client.chat.postMessage({ channel: channelId, text: `Task created: *${title}*`, blocks: buildTaskCard(task), }); // Refresh Home tab for both users await refreshHomeTab(createdBy, client); if (assignee !== createdBy) await refreshHomeTab(assignee, client); }); };

const FRIENDLY = { 'channel_not_found': 'That channel no longer exists.', 'not_in_channel': 'Invite @TeamAssist to this channel first.', 'user_not_found': 'That user could not be found.', 'time_in_past': 'Please choose a future time for your reminder.', 'cant_invite_self': 'You cannot assign yourself as approver.', 'ratelimited': 'Too many requests. Please wait a moment and try again.', }; async function safeCall(fn, respond) { try { return await fn(); } catch (err) { const code = err.data?.error; const msg = FRIENDLY[code] || 'Something went wrong.
