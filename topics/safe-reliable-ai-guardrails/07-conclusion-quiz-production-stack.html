The course’s lasting lesson is a repeatable method: identify the failure, detect it with a validator, correct it with an explicit policy.

Core idea: Compose cheap, independent guards around the model and test them like any other code.

Lesson 10

Conclusion

Generative AI is “a fundamentally non-deterministic programming paradigm”. The wrap-up: guardrails are how you bring such applications to production with reliability. The process is the transferable part: identify the problem, build a validator that detects it, take a corrective action. The instructor encourages exploring the open-source Guardrails Hub of pre-built validators and contributing your own.

// ── Mini guardrails core (mirrors the Guardrails AI vocabulary: Validator, Guard, on_fail) ──
class ValidationError extends Error {}
const pass = () => ({ outcome: 'pass' });
const fail = (errorMessage, fixValue) => ({ outcome: 'fail', errorMessage, fixValue });

class Guard {
  constructor(name = 'guard') { this.name = name; this.rules = []; this.log = []; }
  use(validator, onFail = 'exception') { this.rules.push({ validator, onFail }); return this; }
  validate(text) {
    let cur = text;
    for (const { validator, onFail } of this.rules) {
      const r = validator.validate(cur);
      this.log.push({ validator: validator.constructor.name, outcome: r.outcome, onFail });
      if (r.outcome === 'pass') continue;
      if (onFail === 'exception') throw new ValidationError(`Validation failed for ${this.name}: ${r.errorMessage}`);
      if (onFail === 'fix' && r.fixValue !== undefined) cur = r.fixValue;   // replace and continue
      else if (onFail === 'filter') return null;                              // drop the whole value
      // 'noop': record the failure, keep the value
    }
    return cur;
  }
}

// Hallucination = "not grounded in sources I trust". Pipeline from the lesson:
//   split output into sentences -> retrieve top-k relevant sources per sentence (embeddings + cosine)
//   -> NLI(premise=source, hypothesis=sentence) -> fail if any sentence is not entailed.
// Stand-ins: hashed bag-of-words "embedding" and a lexical NLI, so this runs with zero downloads.
// In the course these are all-MiniLM-L6-v2 and a fine-tuned NLI "provenance" model.
const STOP = new Set('a an the is are was were of in on at to and or for it its this that with by be as from'.split(' '));
const NEG = new Set(['not', 'no', 'never', 'cannot', "isn't", "doesn't", "don't"]);
const toks = s => (s.toLowerCase().match(/[a-z']+/g) || []);
const content = s => toks(s).filter(w => !STOP.has(w) && !NEG.has(w)).map(w => w.replace(/(es|s)$/, ''));

function embed(text, dim = 64) {                         // hashed bag-of-words, L2-normalised
  const v = new Float64Array(dim);
  for (const w of content(text)) { let h = 0; for (const c of w) h = (h * 31 + c.charCodeAt(0)) >>> 0; v[h % dim] += 1; }
  const n = Math.hypot(...v) || 1; return v.map(x => x / n);
}
const cosine = (a, b) => a.reduce((s, x, i) => s + x * b[i], 0);

function nli(premise, hypothesis) {                      // -> 'entailment' | 'contradiction' | 'neutral'
  const p = new Set(content(premise)), h = content(hypothesis);
  const covered = h.filter(w => p.has(w)).length / (h.length || 1);
  const negP = toks(premise).some(w => NEG.has(w)), negH = toks(hypothesis).some(w => NEG.has(w));
  if (covered >= 0.8 && negP !== negH) return 'contradiction';
  return covered >= 0.8 ? 'entailment' : 'neutral';
}

class HallucinationValidator {
  constructor(sources, { topK = 5, embedFn = embed, nliFn = nli } = {}) {
    this.sources = sources; this.topK = topK; this.embedFn = embedFn; this.nliFn = nliFn;
    this.sourceVecs = sources.map(s => embedFn(s));      // pre-compute once, not per request
  }
  sentenceSplitter(text) { return text.split(/(?<=[.!?])\s+/).filter(Boolean); }
  findRelevantSources(sentence) {
    const q = this.embedFn(sentence);
    return this.sources.map((s, i) => ({ s, sim: cosine(q, this.sourceVecs[i]) }))
      .sort((a, b) => b.sim - a.sim).slice(0, this.topK).map(x => x.s);
  }
  checkEntailment(premise, hypothesis) { return this.nliFn(premise, hypothesis) === 'entailment'; }
  validate(text) {
    const hallucinated = this.sentenceSplitter(text).filter(sent =>
      !this.findRelevantSources(sent).some(src => this.checkEntailment(src, sent)));
    return hallucinated.length ? fail(`Hallucinated content: ${JSON.stringify(hallucinated)}`) : pass();
  }
}

// Zero-shot topic detection, the same trick as facebook/bart-large-mnli:
//   premise = the text, hypothesis = "This text is about <topic>", read off P(entailment) per topic.
// Stand-in scorer: keyword evidence squashed to (0,1). Swap `scoreTopics` for a real classifier in production.
const LEXICON = {
  pizza:       ['pizza', 'crust', 'topping', 'pepperoni', 'margherita', 'delivery', 'menu', 'order'],
  politics:    ['election', 'president', 'senate', 'vote', 'party', 'government', 'congress', 'policy'],
  automobiles: ['ford', 'f-150', 'truck', 'sedan', 'engine', 'horsepower', 'car', 'ranger', 'suv'],
  medicine:    ['diagnosis', 'dosage', 'prescription', 'symptom', 'treatment'],
};
function scoreTopics(text, topics) {
  const t = text.toLowerCase();
  return Object.fromEntries(topics.map(topic => {
    const hits = (LEXICON[topic] || []).filter(k => t.includes(k)).length;
    return [topic, 1 - Math.exp(-1.6 * hits)];                 // 0 hits -> 0, 1 -> .80, 2 -> .96
  }));
}
function detectTopics(text, topics, threshold = 0.5) {
  const s = scoreTopics(text, topics);
  return topics.filter(tp => s[tp] > threshold);
}
class ConstrainTopics {
  constructor(bannedTopics, threshold = 0.5) { this.banned = bannedTopics; this.threshold = threshold; }
  validate(text) {
    const hit = detectTopics(text, this.banned, this.threshold);
    return hit.length ? fail(`Banned topics detected: ${hit.join(', ')}`) : pass();
  }
}

// Presidio's shape: an ANALYZER finds entities {type,start,end,score}; an ANONYMIZER rewrites the text.
// Stand-in recognisers: regex + Luhn checksum + a "my name is / Mr." context rule for PERSON.
const luhn = d => { let s = 0, alt = false; for (let i = d.length - 1; i >= 0; i--) { let n = +d[i]; if (alt) { n *= 2; if (n > 9) n -= 9; } s += n; alt = !alt; } return s % 10 === 0; };
const RECOGNISERS = [
  { type: 'EMAIL_ADDRESS', re: /[\w.+-]+@[\w-]+(?:\.[\w-]+)+/g },
  { type: 'US_SSN',        re: /\b\d{3}-\d{2}-\d{4}\b/g },
  { type: 'CREDIT_CARD',   re: /\b(?:\d[ -]?){13,16}\b/g, check: m => luhn(m.replace(/\D/g, '')) },
  { type: 'PHONE_NUMBER',  re: /(?<![\d-])(?:\+?1[ -]?)?(?:\(\d{3}\)|\d{3})[ -]?\d{3}[ -]?\d{4}\b/g },
  { type: 'PERSON',        re: /(?<=\b(?:my name is|i am|i'm|this is|Mr\.|Mrs\.|Ms\.|Dr\.)\s)[A-Z][a-z]+(?:\s[A-Z][a-z]+)?/g },
];
function analyze(text) {
  const found = [];
  for (const { type, re, check } of RECOGNISERS)
    for (const m of text.matchAll(re))
      if (!check || check(m[0])) found.push({ type, start: m.index, end: m.index + m[0].length });
  found.sort((a, b) => a.start - b.start);
  return found.filter((e, i) => i === 0 || e.start >= found[i - 1].end);  // drop overlaps
}
function anonymize(text, entities) {
  let out = '', pos = 0;
  for (const e of entities) { out += text.slice(pos, e.start) + `<${e.type}>`; pos = e.end; }
  return out + text.slice(pos);
}
class DetectPII {
  constructor(entityTypes) { this.types = new Set(entityTypes); }
  validate(text) {
    const hits = analyze(text).filter(e => this.types.has(e.type));
    return hits.length
      ? fail(`PII detected: ${[...new Set(hits.map(h => h.type))].join(', ')}`, anonymize(text, hits))
      : pass();
  }
}
// Output side: stream chunks, but only release text once a full sentence is buffered and validated.
async function* guardStream(chunks, guard) {
  let buf = '';
  for await (const c of chunks) {
    buf += c;
    const parts = buf.split(/(?<=[.!?])(?<!\b(?:Mr|Mrs|Ms|Dr)\.)\s+/);   // don't split after "Mr."
    buf = parts.pop();                                   // keep the unfinished sentence buffered
    for (const sentence of parts) yield guard.validate(sentence) + ' ';
  }
  if (buf) yield guard.validate(buf);
}

// Cascade, cheapest first: (1) exact match  (2) entity extraction  (3) embedding similarity.
// Stand-ins: capitalised-phrase "NER" (the course uses a BERT NER model) and character-trigram vectors
// (the course uses sentence-transformers). Structure and thresholds work the same way.
const trigrams = s => { const t = ` ${s.toLowerCase().replace(/[^a-z ]/g, '')} `; const m = new Map(); for (let i = 0; i < t.length - 2; i++) m.set(t.slice(i, i + 3), (m.get(t.slice(i, i + 3)) || 0) + 1); return m; };
const cos = (a, b) => { let d = 0, na = 0, nb = 0; for (const [k, v] of a) { na += v * v; d += v * (b.get(k) || 0); } for (const v of b.values()) nb += v * v; return d / Math.sqrt(na * nb || 1); };

function extractOrgs(text) {                                  // "Pizza by Alfredo", "Domino's", "Papa Johns"
  const re = /\b[A-Z][\w']*(?:\s(?:by|of|&|and)?\s?[A-Z][\w']*)*/g;
  return [...text.matchAll(re)].map(m => m[0]).filter(s => s.includes(' ') || /'s$|s$/.test(s));
}
class CompetitorCheck {
  constructor(competitors, { threshold = 0.6, ownBrands = [] } = {}) {
    this.competitors = competitors; this.threshold = threshold; this.ownBrands = ownBrands;
    this.vecs = competitors.map(c => trigrams(c));            // pre-computed once
  }
  exactMatch(text) { return this.competitors.filter(c => new RegExp(`\\b${c.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')}\\b`, 'i').test(text)); }
  fuzzyMatch(text) {
    const hits = new Set();
    for (const ent of extractOrgs(text)) { const v = trigrams(ent);
      this.vecs.forEach((cv, i) => { if (cos(v, cv) >= this.threshold) hits.add(this.competitors[i]); }); }
    return [...hits];
  }
  validate(raw) {
    // Our own brand is lexically close to a competitor ("Alfredo's Pizza Cafe" vs "Pizza by Alfredo"): remove it first.
    const text = this.ownBrands.reduce((t, b) => t.split(b).join(' '), raw);
    const exact = this.exactMatch(text);
    if (exact.length) return fail(`Competitor mentioned (exact): ${exact.join(', ')}`);
    const fuzzy = this.fuzzyMatch(text);
    return fuzzy.length ? fail(`Competitor mentioned (similar): ${fuzzy.join(', ')}`) : pass();
  }
}
// The course's four failure modes, closed with four guards around one chatbot.
//   input  guard : ConstrainTopics (unintended use) + DetectPII (leakage to the LLM provider)
//   output guard : HallucinationValidator (groundedness) + CompetitorCheck (reputation) + DetectPII (leakage to the user)
const KB = ["Alfredo's Pizza Cafe opens at 11am daily.", 'Delivery is free for orders over 25 USD.', 'Gluten-free crust costs 3 USD extra.'];
let lastPrompt = '';
const llm = q => (lastPrompt = q, reply(q));
const reply = q =>                                            // deterministic stand-in for the model
  /veggie supreme/i.test(q) ? 'The Veggie Supreme has bell peppers. It bakes for 12 minutes.' :
  /better than/i.test(q)    ? "Alfredo's Pizza Cafe beats Pizza by Alfredo easily." :
  /open/i.test(q)           ? "Alfredo's Pizza Cafe opens at 11am daily." :
  /deliver/i.test(q)        ? 'Delivery is free for orders over 25 USD.' : 'Happy to help!';

const inputGuard = new Guard('input')
  .use(new ConstrainTopics(['politics', 'automobiles']), 'exception')
  .use(new DetectPII(['PERSON', 'PHONE_NUMBER', 'EMAIL_ADDRESS', 'US_SSN', 'CREDIT_CARD']), 'fix');
const outputGuard = new Guard('output')
  .use(new CompetitorCheck(['Pizza by Alfredo'], { ownBrands: ["Alfredo's Pizza Cafe"] }), 'exception') // cheap checks first
  .use(new DetectPII(['PHONE_NUMBER', 'EMAIL_ADDRESS']), 'fix')
  .use(new HallucinationValidator(KB), 'exception');                                                    // model-based check last

function answer(q) {
  try { return { ok: true, text: outputGuard.validate(llm(inputGuard.validate(q))) }; }
  catch (e) { return { ok: false, text: e.message.replace(/^Validation failed for /, '').slice(0, 78) }; }
}
const redTeam = [
  ['control',      'When do you open?'],
  ['control',      'Is delivery free?'],
  ['hallucination','Give me the Veggie Supreme recipe'],
  ['off-topic',    'Compare a Ford F-150 and a Ranger'],
  ['PII in',       'I am Mr. Tate, email me at hank@example.com about delivery'],
  ['competitor',   'Is your pizza better than Pizza by Alfredo?'],
];
console.log('class         | verdict | detail');
for (const [cls, q] of redTeam) { const r = answer(q); console.log(`${cls.padEnd(13)} | ${(r.ok ? 'served' : 'BLOCKED').padEnd(7)} | ${r.text}`); }
console.log('\nWhat the LLM provider saw for the PII request:', lastPromptFor('I am Mr. Tate, email me at hank@example.com about delivery'));
function lastPromptFor(q) { answer(q); return lastPrompt; }

Senior notes

Putting it together

  • Order guards by cost and certainty: regex and lists, then small classifiers, then NLI, then LLM judges. The output guard above runs groundedness last.
  • The red-team table is the deliverable. Keep adversarial cases per failure mode and run them in CI; each guard change must keep the table green.
  • Observe in production. Log guard outcomes, block rates per validator and latency per guard; sudden changes mean drift or attack.
  • Latency budget. Every guard adds time; run independent guards in parallel and stream where possible.
  • Defence in depth, not belief. Guards reduce risk; they do not prove safety. Pair them with least-privilege tool access and human escalation paths.

Frequently Asked Questions

Which guardrails library should I use?

The course uses Guardrails AI, which has a hub of pre-built validators; the concepts also map to other frameworks and to a hand-rolled middleware layer.

What next?

Add guards one failure mode at a time, build the red-team table, and review block-rate dashboards weekly.

Key Takeaway

Guardrails turn a probabilistic component into a system you can reason about: name the failure, detect it, act, and keep the proof in a test.
